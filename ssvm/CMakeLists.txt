cmake_minimum_required(VERSION 3.10)
project(ssvm CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(ssvm STATIC src/parser.cpp src/compiler.cpp src/vm.cpp src/builtins.cpp src/bridge.cpp)
target_include_directories(ssvm PUBLIC include)
add_executable(ssvm_cli src/main.cpp)
target_link_libraries(ssvm_cli ssvm)
set_target_properties(ssvm_cli PROPERTIES OUTPUT_NAME ssvm)
